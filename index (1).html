Library Management System

Python · SQLite

Add books, issue them to members with a due date, and calculate late fees on return. Built to learn clean database design and real-world rules.

Open the code 
import sqlite3
from datetime import date, timedelta

db = sqlite3.connect("library.db")
db.execute("""CREATE TABLE IF NOT EXISTS books(
    id INTEGER PRIMARY KEY, title TEXT, author TEXT,
    issued_to TEXT, due TEXT)""")

def add_book(title, author):
    db.execute("INSERT INTO books(title, author) VALUES (?, ?)",
               (title, author))
    db.commit()

def issue_book(book_id, member, days=14):
    due = (date.today() + timedelta(days=days)).isoformat()
    cur = db.execute(
        "UPDATE books SET issued_to=?, due=? "
        "WHERE id=? AND issued_to IS NULL", (member, due, book_id))
    db.commit()
    return cur.rowcount == 1

def return_book(book_id):
    row = db.execute("SELECT due FROM books WHERE id=? "
                     "AND issued_to IS NOT NULL", (book_id,)).fetchone()
    if not row:
        return 0
    late = max(0, (date.today() - date.fromisoformat(row[0])).days)
    db.execute("UPDATE books SET issued_to=NULL, due=NULL WHERE id=?",
               (book_id,))
    db.commit()
    return late * 2  # fine in rupees

Decentralized Voting DApp

Solidity · Ethers.js

A smart contract where each wallet votes once and results live on-chain, so nobody can quietly change them.

Open the code 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract Voting {
    string[] public candidates;
    mapping(uint => uint) public votes;
    mapping(address => bool) public hasVoted;

    constructor(string[] memory names) {
        candidates = names;
    }

    function vote(uint id) external {
        require(!hasVoted[msg.sender], "Already voted");
        require(id < candidates.length, "Bad candidate");
        hasVoted[msg.sender] = true;
        votes[id]++;
    }

    function winner() external view returns (string memory) {
        uint best;
        for (uint i = 1; i < candidates.length; i++) {
            if (votes[i] > votes[best]) best = i;
        }
        return candidates[best];
    }
}
